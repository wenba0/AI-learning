### git
```
# 开发习惯
git remote add center https://github.com/vllm-projects/vllm.git
git fetch center
git checkout -b bugfix1 center/master
git add .
git commit -a -m "xxxx"
git commit -a --amend --no-edit
git push -u origin bugfix
git push -f

#远端代码更新了，重新拉取
git stash
git pull --rebase origin master
git stash pop

#处理冲突
git fetch center
git checkout -b new center/master #切到其他分支拉取最新代码
git checkout bugfix #切回当前开发分支
git merge new #将new分支合并过来，解冲突

```
